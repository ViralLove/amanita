# JavaScript/Node.js
cat << 'EOF' > validate_example.js
const { validateComponent } = require('./scripts/validators/validate_component_upload.js');

async function checkBeforeAction444() {
  const components = ['amanita_muscaria', 'blue_lotus', 'lions_mane'];
  
  for (const comp of components) {
    const report = await validateComponent(comp, 'localhost');
    
    if (!report.contract.action444_compatible) {
      console.error(`❌ ${comp} not ready for Action 444`);
      process.exit(1);
    }
    
    console.log(`✅ ${comp} validated (score: ${report.qualityScore}/10)`);
  }
  
  console.log('✅ All components ready - safe to run Action 444');
}

checkBeforeAction444();
EOF

node validate_example.js

# Monitor component uploads in real-time
while true; do
  clear
  echo "=== Component Validation Status ($(date +%H:%M:%S)) ==="
  for comp in data/components/*/; do
    component=$(basename "$comp")
    [[ "$component" == "component_features_presets.json" ]] && continue
    score=$(node scripts/validators/validate_component_upload.js --component "$component" --network localhost 2>&1 \
      | grep "Quality Score" | awk '{print $3}')
    echo "$component: $score"
  done
  sleep 5
done
